// Core Business Transaction Dispatcher
async function dispatchTransaction(order: OrderPayload) {
  await prisma.$transaction([
    prisma.stock.decrement({ where: { id: order.sku } }),
    prisma.auditLog.create({ data: { status: 'COMMITTED' } }),
    prisma.invoice.generate({ data: { orderId: order.id } })
  ]);
}
